cmake_minimum_required(VERSION 3.16)
project(digimon_hyper_colosseum_2002 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /EHsc /utf-8)
endif()
set(CARDS_CSV "${CMAKE_CURRENT_SOURCE_DIR}/data/starter_cards.csv")
if(NOT EXISTS "${CARDS_CSV}")
  message(FATAL_ERROR "Missing Starter 1 and 2 catalog: ${CARDS_CSV}")
endif()
add_library(hc_core STATIC src/engine.cpp src/card_catalog.cpp src/auto_player.cpp src/starter_decks.cpp)
target_include_directories(hc_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(hc_cards src/main.cpp)
target_link_libraries(hc_cards PRIVATE hc_core)
if(WIN32)
  file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" HC_SOURCE_PATH)
  file(TO_CMAKE_PATH "${CARDS_CSV}" HC_CSV_PATH)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
  configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/include/card_viewer_paths.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/generated/card_viewer_paths.h" @ONLY)
  add_executable(hc_card_viewer WIN32 src/card_viewer_win32.cpp)
  target_link_libraries(hc_card_viewer PRIVATE hc_core gdiplus)
  target_include_directories(hc_card_viewer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  set_target_properties(hc_card_viewer PROPERTIES
    VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endif()
add_executable(hc_core_tests tests/core_tests.cpp)
target_link_libraries(hc_core_tests PRIVATE hc_core)
add_executable(hc_auto_tests tests/auto_tests.cpp)
target_link_libraries(hc_auto_tests PRIVATE hc_core)
if(WIN32)
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT hc_card_viewer)
else()
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT hc_cards)
endif()
set_target_properties(hc_cards PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
  VS_DEBUGGER_COMMAND_ARGUMENTS "${CARDS_CSV} --demo")
include(CTest)
if(BUILD_TESTING)
  add_test(NAME catalog_and_battle COMMAND hc_cards "${CARDS_CSV}" --demo)
  add_test(NAME starter_card COMMAND hc_cards "${CARDS_CSV}" --card St-2)
  add_test(NAME full_core_tests COMMAND hc_core_tests "${CARDS_CSV}")
  add_test(NAME automatic_duel COMMAND hc_auto_tests)
endif()
